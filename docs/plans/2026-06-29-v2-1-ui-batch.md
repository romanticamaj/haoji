### Task 3: Tap-to-edit speaker list (A2)

**Files:**
- Modify: `app/screens/TranscriptTab.tsx`
- Test: `app/screens/__tests__/TranscriptTab.speakerlist.test.tsx`

**Interfaces:**
- Consumes: `session.transcript.speakers`, `session.names`, `onUpdate`, `nameOf` from `lib/transcript`.
- Produces: a speaker list where each row shows a colour dot + the speaker label + current name (or a faint "點此命名" hint). Tapping a row toggles an inline `TextInput` (autofocus) for that speaker; editing calls `onUpdate` with the updated `names`. Only one row's editor is open at a time (local `editing` state).

- [ ] **Step 1: Write the failing test** `app/screens/__tests__/TranscriptTab.speakerlist.test.tsx`

```tsx
import React from 'react';
import { render, fireEvent } from '@testing-library/react-native';
import TranscriptTab from '../TranscriptTab';
import { SavedSession } from '../../lib/store';

const s: SavedSession = { id: 's', createdAt: 1, title: 't', unlocked: false,
  names: { SPEAKER_00: '' },
  transcript: { speakers: ['SPEAKER_00', 'SPEAKER_01'], segments: [
    { start: 0, end: 2, speaker: 'SPEAKER_00', text: '你好' },
    { start: 2, end: 4, speaker: 'SPEAKER_01', text: '嗨' },
  ] } };

test('tapping a speaker row reveals an input; editing updates names', async () => {
  const onUpdate = jest.fn();
  const { getByText, getByTestId, queryByTestId } = await render(<TranscriptTab session={s} onUpdate={onUpdate} />);
  // editor is not open initially
  expect(queryByTestId('speaker-input-SPEAKER_00')).toBeNull();
  // tap the row for SPEAKER_00
  fireEvent.press(getByTestId('speaker-row-SPEAKER_00'));
  const input = getByTestId('speaker-input-SPEAKER_00');
  fireEvent.changeText(input, 'Gary');
  expect(onUpdate).toHaveBeenCalledWith(expect.objectContaining({ names: expect.objectContaining({ SPEAKER_00: 'Gary' }) }));
});

test('unnamed speaker shows the 點此命名 hint', async () => {
  const { getByText } = await render(<TranscriptTab session={s} onUpdate={() => {}} />);
  getByText('點此命名');
});
```

  const setName = (sp: string, name: string) => onUpdate({ ...session, names: { ...session.names, [sp]: name } });

          onPress={() => setEditing(editing === sp ? null : sp)}>

      </View>
    ))}
  </View>
